2. Server IP Address Validation

For added security, we validate that requests are made from an authorized server. Why We Need Your IP Address:
  • To prevent unauthorized access.
  • This ensures that requests are only accepted from trusted sources
  • To avoid potential threats.
  • ⚠️NB: During the process of using this API tool, any request will be verified by IP address of the server you are using. Only requests coming from the registered IP address will be accepted. so ensure to contact us and validate you IP server adress.

import requests
def make_api_request(endpoint, method, data, api_key):
    url = f'http://localhost:3000/api/{endpoint}'
    headers = {'Content-Type': 'application/json'}
    try:
        response = requests.request(method, url, json=data, headers=headers, params={'api_key': api_key})
        response.raise_for_status()
        return response.json()
    except requests.exceptions.RequestException as err:
        print(f"Error in {method} request to {endpoint}: {err}")
        raise
   
def update_employer(api_key, data):
    try:
        result = make_api_request('employer', 'POST', data, api_key)
        print(result)
    except Exception as error:
        print(f"Error updating employer data: {error}")



public class ApiClient {
    private static final HttpClient client = HttpClient.newHttpClient();
    private static final ObjectMapper objectMapper = new ObjectMapper();
    public static String makeApiRequest(String endpoint, String method, Map data, String apiKey) throws Exception {
    String url = "http://localhost:3000/api/" + endpoint + "?api_key=" + apiKey;
    String requestBody = objectMapper.writeValueAsString(data);
    HttpRequest.Builder requestBuilder = HttpRequest.newBuilder()
        .uri(URI.create(url))
        .header("Content-Type", "application/json");
   
        if (method.equalsIgnoreCase("POST")) {
            requestBuilder.POST(HttpRequest.BodyPublishers.ofString(requestBody));
        }
   
        HttpRequest request = requestBuilder.build();
        try {
                HttpResponse response = client.send(request, HttpResponse.BodyHandlers.ofString());
                return response.body();
            } catch (Exception e) {
                System.err.println("Error in " + method + " request to " + endpoint + ": " + e.getMessage());
                throw e;
            }
    }
   
    public static void updateEmployer(String apiKey, Map data) {
    try {
            String result = makeApiRequest("employer", "POST", data, apiKey);
            System.out.println(result);
        } catch (Exception e) {
            System.err.println("Error updating employer data: " + e.getMessage());
        }
    }
}
